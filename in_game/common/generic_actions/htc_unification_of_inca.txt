htc_uoi_settle_province = {
    type = situation
    show_message = no
    potential = {
        scope:actor = {
            religion = religion:inti
        }
    }

    allow = {
        scope:actor = { at_war = no }
    }

    automation_tick = never
    automation_tick_frequency = 12
    ai_tick = monthly
    ai_tick_frequency = 6

    cooldown = {
        type = htc_uoi_settle_province_key
        years = 5
    }

    select_trigger = {
        looking_for_a = situation
        interaction_source_list = {
            situation:unification_of_inca = {
                add_to_list = source
            }
        }
        target_flag = recipient
        name = "choose_situation"
        column = {
            data = name
        }
        visible = {
            situation:unification_of_inca = this
            situation_is_active = yes
        }
    }

    select_trigger = {
        looking_for_a = province_definition
        interaction_source_list = {
            region:andes_region = {
                every_area_in_region = {
                    every_province_definition_in_area = {
                        add_to_list = source
                    }
                }
			}
		}
        cache_interaction_source_list = yes
        target_flag = target
        name = "choose_province"
        none_available_msg_key = "no_provinces_available"
        column = { data = name }
        column = { data = population }
        visible = {
            any_location_in_province_definition = {
                is_ownable = yes
                has_owner = no
			}
        }
        enabled = {
            any_location_in_province_definition = {
                is_ownable = yes
                has_owner = no
                adjacent_to_owned_or_owned_by_subject = scope:actor
            }
        }
    }
    effect = {
        if = {
            limit = { exists = scope:target }
            scope:actor = {
                add_gold = {
                    value = 10
                    scope:target = {
                        every_location_in_province_definition = {
                            add = development
                        }
                    }
                    max = {
                        value = scope:actor.country_tax_base
                        multiply = 10
                    }
                    multiply = -1
                }
            }
        }
        else = {
            custom_tooltip = {
                text = htc_uoi_settle_province_effect_tt
            }
        }
        scope:target ?= {
            while = {
                limit = {
                    any_location_in_province_definition = {
                        is_ownable = yes
                        has_owner = no
                        adjacent_to_owned_or_owned_by_subject = scope:actor
                    }
                }
                every_location_in_province_definition = {
                    limit = {
                        is_ownable = yes
                        has_owner = no
                        adjacent_to_owned_or_owned_by_subject = scope:actor
                    }
                    change_location_owner = scope:actor
                    change_integration_level = colonized
                }
            }
            # custom_tooltip = {
            #     text = htc_uoi_purchase_province_tt
            #     scope:target = {
            #         trigger_event_non_silently = htc_unification_of_inca.4
            #     }
            # }
        }
    }

    ai_will_do = {
        value = -1
        if = {
            # has more than 3x the gold required to settle
            limit = {
                scope:actor = {    
                    gold > {
                        value = 10
                        scope:target = {
                            every_location_in_province_definition = {
                                add = development
                            }
                        }
                        max = {
                            value = scope:actor.country_tax_base
                            multiply = 10
                        }
                        multiply = 3
                    }
                }
            }
        }
    }
}

htc_uoi_conquer_neighbor = {
    type = situation
    show_message = no
    potential = {
        scope:actor = { religion = religion:inti }
    }

    allow = {
        scope:actor = { at_war = no }
    }

    automation_tick = never
    automation_tick_frequency = 12
    ai_tick = monthly
    ai_tick_frequency = 6

    cooldown = {
        type = htc_uoi_conquer_neighbor_key
        years = 2
    }

    select_trigger = {
        looking_for_a = situation
        interaction_source_list = {
            situation:unification_of_inca = {
                add_to_list = source
            }
        }
        target_flag = recipient
        name = "choose_situation"
        column = {
            data = name
        }
        visible = {
            situation:unification_of_inca = this
            situation_is_active = yes
        }
    }

    select_trigger = {
        looking_for_a = country
        target_flag = target
        name = "choose_country"
        column = { data = name }
        column = { data = population }
        column = { data = expected_army_size }
        visible = {
            any_owned_location = { region = region:andes_region }
        }
        enabled = {
			is_neighbor_of = scope:actor
        }
    }
	
    effect = {
        scope:actor = {
            declare_war_with_cb = {
                target = scope:target
                type = casus_belli:cb_htc_unify_inca
            }
        }
    }

    ai_will_do = {
        value = -1
        if = {
            limit = { current_year > 1340 }
            scope:actor = {
                add = country_strength
                every_country_that_can_be_called_offensively = {
                    add = country_strength
                }
            }
            scope:target = {
                subtract = country_strength
                every_country_that_can_be_called_defensively = {
                    subtract = country_strength
                }
            }
        }
    }
}
