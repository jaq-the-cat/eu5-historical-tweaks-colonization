htc_coi_south_asia_location_count = 1452

# locations this country itself holds in south asia
htc_coi_own_locations_in_south_asia = {
    every_owned_location = {
        limit = { sub_continent = sub_continent:south_asia }
        add = 1
    }
}

# locations this country holds in south asia, counting its whole subject tree
htc_coi_south_asia_locations_owned = {
    value = htc_coi_own_locations_in_south_asia
    every_subject_or_below = {
        add = htc_coi_own_locations_in_south_asia
    }
}

# the same as a share of the subcontinent
htc_coi_south_asia_share_owned = {
    value = htc_coi_south_asia_locations_owned
    divide = htc_coi_south_asia_location_count
}

# scope:target = location
htc_coi_location_purchase_price = {
    value = 100
    add = {
        value = scope:target.development
        multiply = 2
    }
    max = {
        value = scope:target.owner.country_tax_base
        divide = 4
    }
}

# scope:target = province
htc_coi_province_purchase_price = {
    value = 100
    scope:target = {
        every_location_in_province = {
            add = {
                value = development
                multiply = 2
            }
        }
    }
    max = {
        value = scope:target.owner.country_tax_base
        divide = 2
    }
}

# scope:target = the Indian country being offered protection
htc_coi_protection_subsidy = {
    value = scope:target.country_tax_base
    multiply = 10
}

# scope:actor = the European country sending troops
htc_coi_war_assistance_manpower = {
    value = scope:actor.monthly_manpower
    multiply = 10
}
