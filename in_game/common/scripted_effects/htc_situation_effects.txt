conquest_of_india_recalculate_strongest = {
    situation:conquest_of_india = {
        remove_variable = strongest_indian_power
        remove_variable = strongest_european_power
        set_variable = { name = sip_locations value = 0 }
        set_variable = { name = sep_locations value = 0 }
        set_variable = { name = sip_location_percentage value = 0 }
        set_variable = { name = sep_location_percentage value = 0 }
    }
    # strongest = holds the largest share of south asia, not the biggest army.
    # ordering by the raw count is the same ranking as by share and avoids
    # fixed-point precision loss on small fractions.
    culture_group:indian_group = {
        ordered_country_in_culture_group = {
            limit = {
                is_subject = no
                has_presence_in = sub_continent:south_asia
            }
            order_by = htc_coi_south_asia_locations_owned
            max = 1
            check_range_bounds = no
            situation:conquest_of_india = {
                set_variable = {
                    name = strongest_indian_power
                    value = prev
                }
                set_variable = {
                    name = sip_locations
                    value = prev.htc_coi_south_asia_locations_owned
                }
                set_variable = {
                    name = sip_location_percentage
                    value = prev.htc_coi_south_asia_share_owned
                }
            }
        }
    }
    ordered_country = {
        limit = {
            can_see_situation = situation:conquest_of_india
            is_possible_colonial_empire = yes
            has_presence_in = sub_continent:south_asia
        }
        order_by = htc_coi_south_asia_locations_owned
        max = 1
        check_range_bounds = no
        situation:conquest_of_india = {
            set_variable = {
                name = strongest_european_power
                value = prev
            }
            set_variable = {
                name = sep_locations
                value = prev.htc_coi_south_asia_locations_owned
            }
            set_variable = {
                name = sep_location_percentage
                value = prev.htc_coi_south_asia_share_owned
            }
        }
    }
}

unification_of_inca_recalculate_strongest = {
	religion:inti = {
		ordered_country_in_religion = {
			limit = { is_subject = no }
			order_by = {
				add = { value = military_strength }
				add = { value = monthly_income_trade_and_tax }
			}
			max = 1
			check_range_bounds = no
			situation:unification_of_inca = {
				set_variable = {
					name = strongest_incan
					value = prev
				}
				set_variable = {
					name = strongest_incan_score
					value = {
                        add = prev.military_strength
                        add = prev.monthly_income_trade_and_tax
                    }
				}
			}
		}
	}
}
