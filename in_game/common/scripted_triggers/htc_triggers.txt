is_possible_colonial_empire = {
	custom_tooltip = {
		text = is_possible_colonial_empire_tt
		has_ports = yes
		or = {
			is_ai = no
			has_or_had_tag = POR
			has_or_had_tag = CAS
			has_or_had_tag = SPA
			has_or_had_tag = FRA
			has_or_had_tag = ENG
			has_or_had_tag = GBR
			has_or_had_tag = NED
			has_or_had_tag = ITA
			is_colonial_top_overlord = yes
		}
	}
}

is_possible_indian_colonizer = {
	custom_tooltip = {
		text = is_possible_indian_colonizer_tt
		is_possible_colonial_empire = yes
		capital ?= {
			not = { continent = continent:asia }
		}
	}
}

is_indian_colonial_subject = {
	custom_tooltip = {
		text = is_indian_colonial_subject_tt
		capital ?= { sub_continent = sub_continent:south_asia }
		is_subject = yes
		top_overlord ?= {
			is_possible_indian_colonizer = yes
		}
	}
}


is_indian_native = {
	custom_tooltip = {
		text = is_indian_native_tt
		capital ?= { sub_continent = sub_continent:south_asia }
		is_indian_colonial_subject = no
	}
}
