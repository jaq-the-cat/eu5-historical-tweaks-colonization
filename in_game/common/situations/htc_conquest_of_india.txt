conquest_of_india = {
    monthly_spawn_chance = monthly_spawn_chance_unique
    hint_tag = hint_conquest_of_india
    
    can_start = {
        any_country = {
            capital ?= { continent = continent:europe }
            has_presence_in = sub_continent:south_asia
        }
    }

    can_end = {
        custom_tooltip = {
            text = "htc_coi_europeans_kicked_out_of_india"
            not = {
                any_country = {
                    capital ?= { continent = continent:europe }
                    has_presence_in = sub_continent:south_asia
                }
            }
        }
    }
    
    # root = country. 
    visible = {
        or = {
            has_presence_in = sub_continent:south_asia
            and = {
                is_possible_colonial_empire = yes
                any_country_in_diplomatic_range = {
                    capital ?= { sub_continent = sub_continent:south_asia }
                }
            }
        }
    }

    on_start = {
        set_variable = { name = recalculate_countdown value = 0 }
        conquest_of_india_recalculate_strongest = yes

        # europeans
        every_country = {
            limit = {
                capital ?= { continent = continent:europe }
                or = {
                    has_presence_in = sub_continent:south_asia
                    any_country_in_diplomatic_range = {
                        capital ?= { sub_continent = sub_continent:south_asia }
                    }
                }
            }
            trigger_event_non_silently = htc_conquest_of_india.1
        }

        # indians
        every_country = {
            limit = { culture = { has_culture_group = culture_group:indian_group } }
            trigger_event_non_silently = htc_conquest_of_india.2
        }
    }

    # Recalculating sweeps every location in South Asia twice plus every country twice,
    # so it runs quarterly rather than every month.
    on_monthly = {
        change_variable = { name = recalculate_countdown add = 1 }
        if = {
            limit = { var:recalculate_countdown >= 3 }
            set_variable = { name = recalculate_countdown value = 0 }
            conquest_of_india_recalculate_strongest = yes
        }
    }

    on_ended = {}

    tooltip = {
        # colonizer
        if = {
            limit = {
                owner ?= {
                    capital ?= { continent = continent:europe }
                    has_presence_in = sub_continent:south_asia
                }
            }
            custom_tooltip = htc_coi_colonizer_tt
        }
        # indian
        else_if = {
            limit = {
                owner ?= {
                    culture = { has_culture_group = culture_group:indian_group }
                }
            }
            custom_tooltip = htc_coi_indian_tt
        }
        # subject of colonizer
        else_if = {
            limit = {
                owner ?= {
                    capital ?= { sub_continent = sub_continent:south_asia }
                    is_subject = yes
                    overlord = { capital ?= { continent = continent:europe } }
                }
            }
            custom_tooltip = htc_coi_subject_of_colonizer_tt
        }
        else = {
            custom_tooltip = htc_coi_spectator_tt
        }
    }

    map_color = {
        # native indian powers are grey
        if = {
            limit = {
                sub_continent = sub_continent:south_asia
                owner ?= {
                    top_overlord_or_this ?= {
                        capital ?= { continent = continent:europe }
                    }
                }
            }
            value = owner.top_overlord_or_this.country_color
        }
        else_if = {
            limit = {
                sub_continent = sub_continent:south_asia
            }
            value = color_rank_duchy
        }
    }

    secondary_map_color = {
        if = {
            limit = {
                sub_continent = sub_continent:south_asia
                # the variable is cleared when no independent Indian power is left,
                # so guard against dereferencing a dead country
                exists = situation:conquest_of_india.var:strongest_indian_power
                not = {
                    owner ?= {
                        top_overlord_or_this ?= {
                            capital ?= { continent = continent:europe }
                        }
                    }
                }
            }
            value = situation:conquest_of_india.var:strongest_indian_power.country_color
        }
    }
}
