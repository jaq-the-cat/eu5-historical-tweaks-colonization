namespace = colonial_charter

#scope:target is the Province of the charter
colonial_charter.100 = {
	hide_portraits = yes
	type = country_event	

	title = colonial_charter.100.title
	desc = colonial_charter.100.desc
	outcome = neutral

	immediate = {
		save_scope_as = actor
		#Select the best location for a capital
		if = {
			limit = {
				scope:target = {
					any_location_in_province = {
						is_city = yes
						count = 0
					}
				}
			}
			scope:target = {
				ordered_location_in_province = {
					max = 1
					order_by = best_capital_for_colony
					
					save_scope_as = new_town
				}
			}
			
		}
		
		#Get a colonial nation vassal neighboring the target province
		if = {
			limit = {
				scope:target = {
					any_location_in_province = {
						any_coast_border_location = {
							has_owner = yes
							owner = {
								is_subject_of = ROOT
								is_subject_type = colonial_nation
							}
						}
					}
				}
			}
			scope:target = {
				random_location_in_province = {
					limit = {
						any_coast_border_location = {
							has_owner = yes
							owner = {
								is_subject_of = ROOT
								is_subject_type = colonial_nation
							}
						}
					}
					random_coast_border_location = {
						limit = {
							has_owner = yes
							owner = {
								is_subject_of = ROOT
								is_subject_type = colonial_nation
							}
						}
						owner = { save_scope_as = scope_adjacent_area_subject }
					}
				}
			}
		}

		#Get the most populous colonial nation vassal with capital in the same region
		ordered_subject = {
			limit = {
				is_subject_type = colonial_nation
				capital = { region = scope:target.region }
				NAND = { exists = scope:scope_adjacent_area_subject this = scope:scope_adjacent_area_subject }
			}
			order_by = total_population
			max = 1
			save_scope_as = regional_subject
		}

		#Save the Province Capital for the event image
		scope:target = {
			capital = { save_scope_as = target_location }
		}
	}
	
	option = { #Form a new Colonial Subject
		name = colonial_charter.100.a
		trigger = {
			scope:target = { is_overseas_for_owner = yes }
			or = {
				is_ai = no
				and = {
					is_ai = yes
					not = { exists = scope:regional_subject }
					not = { exists = scope:scope_adjacent_area_subject }
				}
			}
		}
		
		if = {
			limit = {
				exists = scope:new_town
			}
			scope:new_town = {
				change_location_rank_effect = { location_rank = location_rank:town }
			}
		}
		
		scope:target = { 
			create_location_country_from_province = {
				subject_type = subject_type:colonial_nation
			
				hidden_effect = {
					setup_colonial_nation = yes

					if = {
						limit = { capital ?= { region = region:brazil_region } }
						change_country_name = COLONY_BRAZIL
						change_country_adjective = COLONY_BRAZIL_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:colombia_region }
							overlord ?= { has_or_had_tag = CAS }
						}
						change_country_name = COLONY_NEW_GRANADA
						change_country_adjective = COLONY_NEW_GRANADA_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:colombia_region }
						}
						change_country_name = COLONY_COLOMBIA
						change_country_adjective = COLONY_COLOMBIA_ADJ
					}
					else_if = {
						limit = { capital ?= { region = region:andes_region } }
						change_country_name = COLONY_PERU
						change_country_adjective = COLONY_PERU_ADJ
					}
					else_if = {
						limit = { capital ?= { region = region:central_america_region } }
						change_country_name = COLONY_GUATEMALA
						change_country_adjective = COLONY_GUATEMALA_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:caribbean_region }
							overlord ?= { has_or_had_tag = CAS }
						}
						change_country_name = COLONY_CUBA
						change_country_adjective = COLONY_CUBA_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:caribbean_region }
							overlord ?= { has_or_had_tag = FRA }
						}
						change_country_name = COLONY_HAITI
						change_country_adjective = COLONY_HAITI_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:aridoamerica_region }
							overlord ?= { tag = CAS }
						}
						change_country_name = COLONY_NEW_CASTILLE
						change_country_adjective = COLONY_NEW_CASTILLE_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:aridoamerica_region }
							overlord ?= { tag = SPA }
						}
						change_country_name = COLONY_NEW_SPAIN
						change_country_adjective = COLONY_NEW_SPAIN_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:great_plains_region }
							overlord ?= { tag = FRA }
						}
						change_country_name = COLONY_LOUISIANA
						change_country_adjective = COLONY_LOUISIANA_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:canada_region }
							overlord ?= { tag = FRA }
						}
						change_country_name = COLONY_QUEBEC
						change_country_adjective = COLONY_QUEBEC_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:canada_region }
						}
						change_country_name = COLONY_CANADA
						change_country_adjective = COLONY_CANADA_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:alaska_region }
						}
						change_country_name = COLONY_ALASKA
						change_country_adjective = COLONY_ALASKA_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:chaco_region }
							overlord ?= { has_or_had_tag = CAS }
						}
						change_country_name = COLONY_PARAGUAY
						change_country_adjective = COLONY_PARAGUAY_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:la_plata_region }
							overlord ?= { has_or_had_tag = CAS }
						}
						change_country_name = COLONY_LA_PLATA
						change_country_adjective = COLONY_LA_PLATA_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:la_plata_region }
						}
						change_country_name = COLONY_PLATINA
						change_country_adjective = COLONY_PLATINA_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:swahili_coast_region }
							overlord ?= { has_or_had_tag = POR }
						}
						change_country_name = COLONY_MOZAMBIQUE
						change_country_adjective = COLONY_MOZAMBIQUE_ADJ
					}
					else_if = {
						limit = {
							capital ?= { region = region:southern_africa_region }
							overlord ?= { has_or_had_tag = ENG }
						}
						change_country_name = COLONY_SOUTH_AFRICA
						change_country_adjective = COLONY_SOUTH_AFRICA_ADJ
					}
				}
			}
		}
		
		ai_will_select = {
			value = 50
		}
	}

	option = { # Join it to scope_adjacent_area_subject
		name = colonial_charter.100.b

		trigger = {
			exists = scope:scope_adjacent_area_subject
		}

		if = {
			limit = {
				exists = scope:new_town
			}
			scope:new_town = {
				change_location_rank_effect = { location_rank = location_rank:town }
				cc_setup_new_town = yes
			}
		}

		scope:target = {
			change_province_owner = scope:scope_adjacent_area_subject
		}

		ai_will_select = {
			value = 1000
		}
	}

	option = { # Join it to regional_subject
		name = colonial_charter.100.f

		trigger = {
			exists = scope:regional_subject
		}

		if = {
			limit = {
				exists = scope:new_town
			}
			scope:new_town = {
				change_location_rank_effect = { location_rank = location_rank:town }
				cc_setup_new_town = yes
			}
		}

		scope:target = {
			change_province_owner = scope:regional_subject
		}

		ai_will_select = {
			value = 2000
		}
	}

	option = { #Keep it as part of the Metropolis
		name = colonial_charter.100.d
		
		trigger = {
			or = {
				is_ai = no
				scope:target = { is_overseas_for_owner = no }
			}
		}

		scope:target = { 
			every_location_in_province = {
				limit = { integration_level = colonized }
				change_integration_level = integrated
			}
		}

		ai_chance = {
			value = 100
		}
	}

	option = { #Play as the Colony
		name = colonial_charter.100.e
		high_risk_option = yes
		trigger = {
			is_human = yes
			scope:target = { is_overseas_for_owner = yes }
		}
		if = {
			limit = {
				exists = scope:new_town
			}
			scope:new_town = {
				change_location_rank_effect = { location_rank = location_rank:town }
			}
		}
		scope:target = { 
			create_location_country_from_province = {
				subject_type = subject_type:colonial_nation
				hidden_effect = {
					setup_colonial_nation = yes
				}
				save_scope_as = target_country
			}
		}
		custom_tooltip = {
			text = colonial_charter.100.e.tt
			change_player = scope:target_country
		}

		# needed when ai and human is playing together
		ai_chance = {
			value = -1000
		}
	}

	after = {
		trigger_event_silently = treaty_of_tordesillas.1
	}
}
