namespace = htc_merge_colonies

htc_merge_colonies.1 = {
	type = country_event
	title = htc_merge_colonies.title
	desc = htc_merge_colonies.desc

	illustration_tags = {
		10 = happy
		10 = exterior
	}
	
	trigger = {
		is_ai = yes
		is_colonial_subject = yes
		overlord ?= { is_ai = yes }
		capital ?= { exists = yes }
		any_neighbor_country = {
			is_colonial_subject = yes
			overlord = root.overlord
			capital ?= {
				region = root.capital.region
			}
		}
	}

	immediate = {
		random_neighbor_country = {
			limit = {
				is_colonial_subject = yes
				overlord = root.overlord
				capital ?= {
					region = root.capital.region
				}
			}
			save_scope_as = other
		}
	}

	option = {
		name = htc.options.take
		historical_option = yes
		ai_will_do = 1000
		
		annex_country = {
			country = scope:other
			reason = Diplomatic
			transfer = yes
		}

		every_owned_location = {
			change_integration_level = integrated
		}
	}

	option = {
		name = htc.options.no
	}
}
