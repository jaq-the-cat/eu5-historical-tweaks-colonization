# offer protection
# root = indian, scope:actor = colonizer, scope:target = indian
htc_conquest_of_india.5 = {
	type = country_event
	category = situation_event
	outcome = neutral

	title = htc_conquest_of_india.5.title
	desc = htc_conquest_of_india.5.desc

	illustration_tags = {
		10 = interior
		10 = regular
	}

	# accept
	option = {
		name = htc_conquest_of_india.5.a

		add_gold = htc_coi_protection_subsidy
		scope:actor ?= {
			add_gold = {
				value = htc_coi_protection_subsidy
				multiply = -1
			}
		}

		if = {
			limit = { is_subject = yes }
			overlord ?= { cancel_subject = prev }
		}

		make_subject_of = {
			target = scope:actor
			type = subject_type:indian_vassal
		}

		scope:actor ?= {
			trigger_event_non_silently = htc_conquest_of_india.6
		}
	}

	# refuse
	option = {
		name = htc_conquest_of_india.5.b

		scope:actor ?= {
			add_opinion = {
				target = root
				modifier = declined_demands_modifier
			}
			trigger_event_non_silently = htc_conquest_of_india.7
		}
	}
}

		10 = regular
	}

	# accept
	option = {
		name = htc_conquest_of_india.8.a

		add_manpower = {
			value = htc_coi_war_assistance_manpower
		}
		scope:actor ?= {
			add_manpower = {
				value = htc_coi_war_assistance_manpower
				multiply = -1
			}

			take_over_all_wars = root
		}
